    D --> E["Keep eligible records"]
    A --> P["Distinct entities"]
    T["Target reference"] --> J["Expand entities against targets"]
    P --> J
    E --> K["Combine eligible and generated records"]
    J --> K
    K --> L["Remove superseded records"]

    D -.-> Q
    K -.-> Q
    F -.-> Q
